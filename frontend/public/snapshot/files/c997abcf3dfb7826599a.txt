+"""``key=value`` pair parsing."""
+
+from __future__ import annotations
+
+
+class ParseError(ValueError):
+    """Raised when :func:`parse_pairs` cannot parse its input."""
+
+
+def parse_pairs(text: str, sep: str = ";") -> dict[str, str]:
+    """Parse ``"key=value; other=value"`` into a dictionary.
+
+    Segments are separated by ``sep``; empty or whitespace-only segments are
+    ignored. Keys and values are stripped of surrounding whitespace. When a key
+    repeats, the last value wins. Only the first ``=`` in a segment separates the
+    key from the value; the value keeps any later ``=``.
+
+    Args:
+        text: Input text.
+        sep: Segment separator.
+
+    Returns:
+        The parsed pairs.
+
+    Raises:
+        ParseError: If a segment is malformed.
+    """
+    result: dict[str, str] = {}
+    for segment in text.split(sep):
+        stripped_segment = segment.strip()
+        if not stripped_segment:
+            continue
+        if "=" not in stripped_segment:
+            raise ParseError(f"Malformed pair segment: {stripped_segment!r}")
+        key, _, value = stripped_segment.partition("=")
+        key = key.strip()
+        value = value.strip()
+        if not key:
+            raise ParseError(f"Malformed pair segment (empty key): {stripped_segment!r}")
+        result[key] = value
+    return result
